/* 이체: 출금 계좌 → 입금 계좌 (원 단위 정수) */
int transfer(Account *from, Account *to,
             int64_t amount)
{
    if (amount <= 0)            return ERR_AMOUNT;
    if (from->balance < amount) return ERR_BALANCE;
    lock_in_order(from, to);   /* 데드락 방지 */
    from->balance -= amount;
    to->balance   += amount;
    write_txlog("TRF", from, to, amount);
    unlock_both(from, to);
    return OK;
}
